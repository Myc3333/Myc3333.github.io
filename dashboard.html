temperature_plot =
  weather_data |>
  ggplot(aes(x = date, y = tmax_c)) +
  geom_line(color = "steelblue") +
  labs(
    x = "Date",
    y = "Maximum temperature (°C)"
  ) +
  theme_minimal()

Maximum temperature by month

distribution_plot =
  weather_data |>
  filter(!is.na(tmax_c)) |>
  ggplot(aes(x = month, y = tmax_c)) +
  geom_boxplot(
    fill = "lightblue",
    outlier.size = 1
  ) +
  labs(
    x = "Month",
    y = "Maximum temperature (°C)"
  ) +
  theme_minimal()

ggplotly(distribution_plot)

Recorded snowfall days by month

snow_summary =
  weather_data |>
  group_by(month) |>
  summarize(
    observed_days = sum(!is.na(snow)),
    snowfall_days = sum(snow > 0, na.rm = TRUE)
  ) |>
  filter(observed_days > 0)

snow_plot =
  snow_summary |>
  ggplot(aes(x = month, y = snowfall_days)) +
  geom_col(fill = "steelblue") +
  labs(
    x = "Month",
    y = "Days with recorded snowfall"
  ) +
  theme_minimal()

ggplotly(snow_plot)

Counts include only days with recorded snowfall data. Months with no recorded snowfall data are omitted.
